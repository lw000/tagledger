  -ArgumentList '--port=8080','--data-dir=data','--web-dir=web','--config-dir=config' `
  -WorkingDirectory c:\github\cpp_works\tagledger -PassThru -WindowStyle Hidden
Start-Sleep -Seconds 2
Invoke-WebRequest -UseBasicParsing http://127.0.0.1:8080/index.html -TimeoutSec 5
Stop-Process -Id $p.Id -Force

Copy-Item web\index.html build\web\index.html -Force
Get-Item build\web\index.html | Select-Object Length, LastWriteTime
